总则：统一评分、证据和输出契约

本算法所有分数均使用明确量纲，所有结论都必须能追溯到规则和原始位置。五行“占比”只表示原局力量，不等同于用神概率；喜神也不能机械地定义为“生用神的五行”。本文各步骤的规则、伪代码和输出契约必须保持一致。

五行力量默认按“天干 1.0、地支本气 1.0、中气 0.6、余气 0.3”计，月令藏干乘可配置季节系数（默认 1.5）。合化只有满足得令、有根、无争合、无克破等条件时才增加化神力量，否则只记录“存在合”。强弱五项先分别归一化到 0~100，再按得令 40%、得地 30%、得势 15%、季节 10%、关系 5% 合成并 clamp 到 0~100；等级阈值统一为极弱<25、身弱<40、中和<60、身强<75、极旺≥75。

取用不再采用“调候五行固定加满分”的单线排序。先确认特殊格局，再以月令格局的病、药、相神为主线，同时把调候分成“未见、已见不足、已足、偏多”四态；只有未见或不足时才增加调候候选分，已足只记录为“调候已备”，偏多还要防止继续增印比。喜神是能辅助用神、制约忌神且不使原局更偏的候选；忌神是直接破坏平衡、格局或用神者；仇神是助忌神者。用、喜、忌、仇、闲五种角色必须互斥，同一天干或同一五行不得同时承担两个角色；闲神取确定性的剩余五行，禁止使用 `set.pop()`。

取用体系默认采用**子平法**：月令为纲，先由月令本气建立格局候选，再以相神、成格/破格、调候和日主承载能力交叉校验。必须区分“格局用神”“调候用神”和“扶抑用神”，三者可以同向，也可以并存但主次不同；调候不得绕过格局而直接覆盖所有候选，已经在原局透根且足用的调候字不得再次按缺失状态加满分。特殊格局条件未齐时只输出候选或人工复核。

还必须计算“承载能力”：它回答“某五行理论上有利时，日主和环境是否能让这份作用真正发生”。例如丁火在亥月遇甲木，甲木作为印星有生火方向，但火弱、寒湿、火根受冲时，甲木的“生火”效力会被折减；不能仅凭 `木生火` 就列为高优先级喜神。承载能力只折减**需要日主主动承接、驾驭或转化**的作用（如以木生火、身强任财、食伤泄秀）；调候火的直接温局效果不应被同一系数机械压低。

候选分数必须同时保存 `raw_score`（理论方向分）和 `effective_score`（经过承载条件后的实际分），并在输出中说明折减原因。`percent` 仍只在同角色候选集合内计算，不能把它解释成传统命理的客观概率。

```python
def calc_element_balance(chart, relations, config):
    raw = {e: 0.0 for e in FIVE_ELEMENTS}; sources = {e: [] for e in FIVE_ELEMENTS}
    add_stems(raw, sources, chart.stems, weight=config.stem_weight)
    add_hidden_stems(raw, sources, chart.branches,
                     weights=(1.0, 0.6, 0.3),
                     month_multiplier=config.month_multiplier)
    for relation in relations.values():
        for item in relation:
            if item.get("effective") and item.get("element"):
                raw[item["element"]] += item.get("power", 0)
                sources[item["element"]].append(item.get("evidence", []))
    total = sum(raw.values())
    return {e: {"raw": raw[e], "percent": percent(raw[e], total),
                "sources": sources[e]} for e in FIVE_ELEMENTS}
```

Step 11: 日主强弱判定引擎

这是整个分析引擎的第一块基石，所有后续判断都依赖它。

说明：本节的 `judge_*` 函数只负责采集原始证据；不得把它们返回的分值直接相加。由 `assess_strength()` 将每项归一化到 0~100 后按总则中的五项权重合成，最终分数强制限制在 0~100。

11.2 得令判定（权重40%）

def score_de_ling(chart: dict, relations: dict, config) -> ComponentScore:
    """
    月令藏干按本气/中气/余气权重计算对日主的生扶与克泄耗，
    再映射到0..100。50代表月令对日主生克大致相当，不再返回 +40/-40 与总分直接相加。
    """
    month_branch = chart["month_branch"]
    weights = config.hidden_weights       # 默认 [1.0, 0.6, 0.3]
    support, oppose, evidence = 0.0, 0.0, []
    for hidden, weight in zip(HIDDEN_STEMS[month_branch], weights):
        ten_god = get_shi_shen(chart["day_stem"], hidden)
        effective = weight * relation_multiplier(month_branch, hidden, relations)
        if ten_god in ("比肩", "劫财", "正印", "偏印"):
            support += effective
        else:
            oppose += effective
        evidence.append(Evidence("strength.de_ling", month_branch, ten_god,
                                 effective, f"月令藏干{hidden}"))

    value = clamp(50 + 50 * safe_ratio(support - oppose, support + oppose), 0, 100)
    tomb = detect_tomb_trigger(month_branch, chart["branches"], relations)
    if tomb.triggered:
        evidence.append(Evidence(
            "strength.tomb_trigger", month_branch, tomb.cause, 0,
            f"墓库受{tomb.cause}引动；是吉是凶另看库中何物透藏及喜忌，不自动加分"
        ))
    return ComponentScore(value=round(value, 2), evidence=evidence,
                          meta={"tomb": tomb})

@dataclass
class TombTrigger:
    triggered: bool
    cause: str | None          # "冲" / "刑" / "冲刑" / None
    relations: list

def detect_tomb_trigger(tomb_branch, four_branches, relations) -> TombTrigger:
    clash_items = find_relations(relations, kind="冲", target=tomb_branch)
    punishment_items = find_relations(relations, kind="刑", target=tomb_branch)
    cause = ("冲刑" if clash_items and punishment_items else
             "冲" if clash_items else "刑" if punishment_items else None)
    return TombTrigger(bool(cause), cause, [*clash_items, *punishment_items])

11.3 得地判定（权重30%）

def score_de_di(chart: dict, relations: dict, kong_wang: dict, config) -> ComponentScore:
    """
    逐一计算日主真根和印星生扶，只计一次藏干力量。
    十二长生仅作语义证据，不再在同一藏干根气上重复加分。
    """
    roots = score_effective_roots(
        chart["day_stem"], chart["branches"], relations, kong_wang,
        hidden_weights=config.hidden_weights,
        root_multiplier=1.0,
        resource_multiplier=config.resource_root_multiplier,
        arching_multiplier=config.arching_root_multiplier,
    )
    value = percent(clamp(roots.total, 0, roots.maximum), roots.maximum)
    return ComponentScore(value=value, evidence=roots.evidence,
                          meta={"real_roots": roots.real,
                                "virtual_roots": roots.arching,
                                "twelve_life": roots.twelve_life_labels})

11.4 得势判定（权重15%）

def score_de_shi(chart: dict, relations: dict) -> ComponentScore:
    """
    天干帮扶/克泄耗均先计算透干的有效力量，再映射到0..100。
    合而不化、合化成功、争合、克制不能都按同一个天干满分计。
    """
    support, oppose, evidence = 0.0, 0.0, []
    for position, stem in enumerate(chart["stems"]):
        if position == 2:                 # 跳过日干自身
            continue
        ten_god = get_shi_shen(chart["day_stem"], stem)
        p = effective_exposed_stem_power(stem, position, chart, relations)
        if ten_god in ("比肩", "劫财", "正印", "偏印"):
            support += p
        else:
            oppose += p
        evidence.append(Evidence("strength.de_shi", stem, ten_god, p,
                                 f"天干位置={position}"))
    value = clamp(50 + 50 * safe_ratio(support - oppose, support + oppose), 0, 100)
    return ComponentScore(value=round(value, 2), evidence=evidence)

11.5 综合打分与等级判定

def assess_strength(chart: dict, relations: dict, kong_wang: dict, config) -> StrengthResult:
    """各组件先归一化到0~100，再按统一权重合成。"""
    ling = score_de_ling(chart, relations, config)                 # 0..100
    di = score_de_di(chart, relations, kong_wang, config)          # 0..100
    shi = score_de_shi(chart, relations)                           # 0..100
    season = score_season_adjustment(chart, config)                # 0..100
    flow = score_relation_adjustment(chart, relations, kong_wang)  # 0..100
    total = clamp(0.40*ling.value + 0.30*di.value + 0.15*shi.value
                  + 0.10*season.value + 0.05*flow.value, 0, 100)
    level = classify_strength(total)
    
    return StrengthResult(
        score=round(total, 2), level=level,
        de_ling=(ling.value >= 50),
        de_di_score=di.value,
        de_shi_score=shi.value,
        components={"de_ling": ling, "de_di": di, "de_shi": shi,
                    "season": season, "relations": flow},
        evidence=flatten_evidence(ling, di, shi, season, flow)
    )

Step 12: 调候需求判定

12.1 核心数据

调候必须采用“日主 × 月令”的具体天干表，不能只凭季节把所有巳午未月统一判成壬癸水，也不能把壬、癸视为可互换。表内天干有先后顺序，例如乙木未月以癸水雨露滋润为先，壬水为次；具体表未录入时只允许输出“调候候选”，不得锁定用神。

调候结果必须返回四态：

- `missing`：所需调候字未见，可按完整调候分进入候选；
- `insufficient`：仅见余气、受冲、落空或受合化，按不足程度折减加分；
- `satisfied`：所需字透干有根且未受损，只记录“调候已备”，不得再次固定加满分；
- `excessive`：同类已偏多或继续生扶会加重格局之病，对同类候选扣分。

def judge_climate(day_stem: str, month_branch: str, four_stems: List[str],
                  four_branches: List[str], relations: list,
                  kong_wang: dict, balance: dict) -> dict:
    """
    返回: {
        "needed": bool,         # 是否需要调候
        "element": str,         # 急需五行
        "stems": list,          # 具体调候天干
        "state": str,           # missing / insufficient / satisfied / excessive
        "present_stems": list,  # 必须记录具体是壬还是癸，不得只记“水”
        "urgency": str,         # "urgent" / "secondary" / "none"
        "reason": str
    }
    """
    info = DAY_MASTER_CLIMATE_TABLE.get((day_stem, month_branch),
                                         CLIMATE_TABLE.get(month_branch, {}))
    
    if info.get("need") is None:
        return {
            "needed": False,
            "element": None,
            "stems": [],
            "state": "not_applicable",
            "present_stems": [],
            "occurrences": [],
            "urgency": "none",
            "reason": info.get("reason", ""),
        }
    
    need_element = info["need"]
    need_stems = info["stems"]
    
    occurrences = []
    for wanted in need_stems:
        occurrences.extend(find_exact_stem_occurrences(
            wanted, four_stems, four_branches, relations, kong_wang
        ))

    # 精确到具体天干：癸透不能证明壬透；壬有根也不能替代癸的透干。
    usable_power = sum(o.effective_power for o in occurrences)
    same_element_percent = balance[STEM_ELEMENT[need_stems[0]]]["percent"]
    if not occurrences:
        state = "missing"
    elif usable_power < CLIMATE_SUFFICIENT_THRESHOLD:
        state = "insufficient"
    elif same_element_percent > CLIMATE_EXCESSIVE_PERCENT:
        state = "excessive"
    else:
        state = "satisfied"
    
    urgency = info.get("priority", "urgent")  # 默认urgent
    
    return {
        "needed": True,
        "element": need_element,
        "stems": need_stems,
        "state": state,
        "present_stems": sorted({o.stem for o in occurrences}),
        "occurrences": occurrences,
        "urgency": urgency,
        "reason": info["reason"],
        "score_policy": {
            "missing": "完整调候分",
            "insufficient": "按有效力量折算",
            "satisfied": "0分，只记录调候已备",
            "excessive": "同类候选扣分"
        }
    }

Step 13: 格局判定引擎

13.0 月令本气与人元司令必须分开

“月支本气取格”不等于“该本气正在分日司令”。若分析器只有四柱、没有出生公历时刻及节气交接时刻，只能输出“未月本气己土，立偏财格候选”，不得写“财星司令”。只有 `birth_context` 包含精确公历时刻、前后节气时刻及选定的人元司令表版本时，才计算分日司令：

```python
def determine_month_command_hidden_stem(birth_context, month_branch, command_table):
    if not birth_context.has_exact_solar_time or not birth_context.has_jieqi_boundary:
        return {"status": "unknown", "reason": "缺少节气交接时刻，不判分日司令"}
    elapsed = birth_context.solar_time - birth_context.previous_jieqi_time
    return command_table.lookup(month_branch, elapsed)
```

结果中同时保留：

- `pattern_basis="month_main_qi"`：按月令本气立格；
- `human_command.status="known/unknown"`：分日司令是否可判；
- `human_command.stem`：已知时才填写司令藏干；
- `command_table_version`：所采用的司令分日表版本，避免不同门派表格混用。

13.1 取格逻辑

def determine_pattern(chart: dict, combos: list, occurrences: list,
                      relations: dict, human_command: dict) -> dict:
    """
    返回: {
        "pattern": str,         # 格局名称
        "pattern_type": str,    # "正格" / "建禄" / "羊刃" / "特殊"
        "cheng_ge": bool,       # 成格/破格
        "supports": list,       # 成格依据、相神与救应
        "conflicts": list,      # 破格原因（如有）
        "pattern_basis": str,   # month_main_qi / exposed_hidden / human_command
        "human_command": dict,
    }
    """
    day_stem = chart["day_stem"]
    month_branch = chart["month_branch"]
    four_stems = chart["stems"]
    four_branches = chart["branches"]

    # 1. 取月令本气，但保留中气、余气及分日司令证据
    ben_qi = HIDDEN_STEMS[month_branch][0]
    
    # 2. 判断是否建禄/羊刃
    # 建禄必须按十干临官位精确判断，不能“与日主同五行就算建禄”。
    JIAN_LU = {"甲":"寅", "乙":"卯", "丙":"巳", "丁":"午", "戊":"巳",
                "己":"午", "庚":"申", "辛":"酉", "壬":"亥", "癸":"子"}
    if month_branch == JIAN_LU[day_stem]:
        return {
            "pattern": "建禄格", "pattern_type": "建禄", "cheng_ge": True,
            "supports": [f"{day_stem}日主临官在{month_branch}"], "conflicts": [],
            "pattern_basis": "month_lu", "human_command": human_command,
            "competing_patterns": [],
        }
    
    if is_yang_blade(day_stem, month_branch):
        return {
            "pattern": "羊刃格", "pattern_type": "羊刃", "cheng_ge": True,
            "supports": [f"{day_stem}日主羊刃在{month_branch}"], "conflicts": [],
            "pattern_basis": "month_blade", "human_command": human_command,
            "competing_patterns": [],
        }
    
    # 3. 正八格：月令本气透干取格
    # 先检查本气是否透干
    ben_qi_shi_shen = get_shi_shen(day_stem, ben_qi)
    
    # 检查月令藏干是否“同一天干”透出；同五行异阴阳不得冒充透干。
    # 同时透出多个藏干时不能只取数组第一个，须全部保留为争格候选。
    hidden_stems = HIDDEN_STEMS[month_branch]
    exposed_hidden = [h for h in hidden_stems
                      if exact_stem_is_exposed(h, chart, exclude_day_stem=True)]
    pattern_stem = select_pattern_stem(
        exposed_hidden=exposed_hidden,
        hidden_stems=hidden_stems,
        human_command=human_command,
        chart=chart,
    )
    shi_shen = get_shi_shen(day_stem, pattern_stem)
    
    PATTERN_MAP = {
        "正官": "正官格", "七杀": "七杀格",
        "正印": "正印格", "偏印": "偏印格",
        "正财": "正财格", "偏财": "偏财格",
        "食神": "食神格", "伤官": "伤官格",
    }
    
    pattern_name = PATTERN_MAP.get(shi_shen, "月劫格" if shi_shen == "劫财" else f"{shi_shen}格")
    
    # 4. 判成格/破格，必须使用带位置、透藏、力量、空亡和合冲状态的十神出现项
    integrity = check_pattern_integrity(
        pattern_name, pattern_stem, occurrences, combos, relations
    )
    basis = "exposed_hidden" if pattern_stem in exposed_hidden else "month_main_qi"
    competing = [PATTERN_MAP.get(get_shi_shen(day_stem, h), f"{get_shi_shen(day_stem, h)}格")
                 for h in exposed_hidden if h != pattern_stem]
    
    return {
        "pattern": pattern_name,
        "pattern_type": "正格",
        "pattern_stem": pattern_stem,
        "cheng_ge": integrity["cheng_ge"],
        "supports": integrity["supports"],
        "conflicts": integrity["conflicts"],
        "pattern_basis": basis,
        "human_command": human_command,
        "competing_patterns": competing,
    }

13.2 成格/破格检验

def check_pattern_integrity(pattern: str, pattern_stem: str,
                            occurrences: list, combos: list,
                            stem_relations: list) -> dict:
    """
    返回成格与破格的双向证据。不得只根据“某十神在四柱出现过”就判定；
    必须检查是否透干、是否贴近、力量是否达阈值、是否空亡或受制。
    """
    supports = []
    conflicts = []

    def effective(ten_gods, min_power=COMBO_MIN_POWER):
        return [o for o in occurrences
                if o.ten_god in ten_gods and o.effective_power >= min_power]
    
    if pattern == "正官格":
        # 破格条件：伤官见官、官被合化、官杀混杂无制
        guan = effective({"正官"})
        shang = effective({"伤官"})
        yin = effective({"正印", "偏印"})
        cai = effective({"正财", "偏财"})
        sha = effective({"七杀"})
        if guan:
            supports.append("正官有效得用")
        if any_combo(shang, guan) and not (yin or cai):
            conflicts.append("伤官见官无通关")
        if guan and sha and not check_guan_sha_resolution(occurrences, stem_relations):
            conflicts.append("官杀混杂无制")
    
    elif pattern == "七杀格":
        # 成格条件：有食神制杀或印化杀
        sha = effective({"七杀"})
        shi = effective({"食神"})
        yin = effective({"正印", "偏印"})
        if any_combo(shi, sha):
            supports.append("食神制杀有效")
        elif any_combo(sha, yin):
            supports.append("杀印相生有效")
        else:
            conflicts.append("杀无制无化")
    
    elif pattern == "食神格":
        # 破格：枭印夺食
        shi = effective({"食神"})
        xiao = effective({"偏印"})
        cai = effective({"正财", "偏财"})
        if any_combo(xiao, shi) and not any_combo(cai, xiao):
            conflicts.append("枭印夺食")
        elif shi:
            supports.append("食神有效且未受枭印破坏")
    
    elif pattern == "伤官格":
        # 成格：伤官配印或伤官生财
        # 破格：伤官见官无通关
        shang = effective({"伤官"})
        guan = effective({"正官"})
        yin = effective({"正印", "偏印"})
        cai = effective({"正财", "偏财"})
        if any_combo(shang, yin):
            supports.append("伤官配印")
        if any_combo(shang, cai):
            supports.append("伤官生财")
        if any_combo(shang, guan) and not (yin or cai):
            conflicts.append("伤官见官无通关")
    
    elif pattern == "正财格" or pattern == "偏财格":
        # 破格：比劫夺财
        wealth = effective({"正财", "偏财"})
        peers = effective({"比肩", "劫财"})
        officer = effective({"正官", "七杀"})
        if wealth:
            supports.append("财星有效得用")
        if sum(o.effective_power for o in peers) >= ROB_WEALTH_POWER_THRESHOLD:
            if officer:
                supports.append("官杀制比护财")
            else:
                conflicts.append("比劫夺财无官杀制")

    return {
        "cheng_ge": len(conflicts) == 0,
        "supports": supports,
        "conflicts": conflicts,
    }

Step 14: 特殊格局判定

特殊格局必须返回 `status`（`confirmed`、`needs_review` 或 `excluded`）及支持/排除证据。条件不完整时只能标记待复核，不能直接改变用神；从格临界分数 24~30 统一标记为 `needs_review`。

14.1 从格判定

def check_cong_ge(chart: dict, strength: StrengthResult,
                  occurrences: list, relations: list,
                  kong_wang: dict) -> dict:
    """
    从格不能只用“日主分低 + 某五行最多”判定。先验日主是否真能弃命，
    再在从财、从杀、从儿三种势中逐格检查。
    """
    day_stem = chart["day_stem"]
    support = effective_power(occurrences, {"比肩", "劫财", "正印", "偏印"})
    real_roots = find_day_master_roots(chart, relations, kong_wang,
                                       include_arching=False)
    unbroken_roots = [r for r in real_roots if r.effective_power >= ROOT_BREAK_THRESHOLD]

    base_evidence = [
        f"日主强弱分={strength.score}",
        f"印比有效力量={support}",
        f"未破真根数={len(unbroken_roots)}",
    ]
    if strength.score > 30 or support > CONG_SUPPORT_MAX or unbroken_roots:
        return {
            "type": "从格", "sub_type": None, "status": "excluded",
            "evidence": base_evidence,
            "conflicts": ["日主未到极弱弃命条件或仍有有效印比根援"],
        }

    family_power = {
        "从财格": effective_power(occurrences, {"正财", "偏财"}),
        "从杀格": effective_power(occurrences, {"正官", "七杀"}),
        "从儿格": effective_power(occurrences, {"食神", "伤官"}),
    }
    subtype = max(family_power, key=family_power.get)
    dominant_power = family_power[subtype]
    second_power = sorted(family_power.values(), reverse=True)[1]

    conflicts = []
    supports = [f"{subtype}主势有效力量={dominant_power}"]
    if dominant_power < CONG_DOMINANT_MIN:
        conflicts.append("所从之势不足")
    if dominant_power - second_power < CONG_DOMINANCE_GAP:
        conflicts.append("财、官杀、食伤多势并见，不能专从")

    if subtype == "从财格":
        # 食伤生财可为顺势；有力官杀泄财过甚则须降级复核。
        if effective_power(occurrences, {"食神", "伤官"}) >= FLOW_MIN:
            supports.append("食伤生财，气势顺行")
        if effective_power(occurrences, {"正官", "七杀"}) > CONG_SIDE_FORCE_MAX:
            conflicts.append("官杀分泄财势过重")
    elif subtype == "从杀格":
        # 财生杀可为顺势；有力食伤制杀即逆势。
        if effective_power(occurrences, {"正财", "偏财"}) >= FLOW_MIN:
            supports.append("财星生杀，气势顺行")
        if any_combo(effective_occurrences(occurrences, {"食神", "伤官"}),
                     effective_occurrences(occurrences, {"七杀"})):
            conflicts.append("有力食伤制杀，逆所从之势")
    elif subtype == "从儿格":
        # 从儿以食伤为主，生财可顺泄；强印夺食、强官杀制身为逆势。
        if effective_power(occurrences, {"正财", "偏财"}) >= FLOW_MIN:
            supports.append("食伤生财，儿又生儿")
        if effective_power(occurrences, {"正印", "偏印"}) > CONG_OPPOSING_MAX:
            conflicts.append("印星有力夺食，不从儿")
        if effective_power(occurrences, {"正官", "七杀"}) > CONG_OPPOSING_MAX:
            conflicts.append("官杀有力，气势不纯")

    if conflicts:
        status = "excluded"
    elif 24 < strength.score <= 30 or support > CONG_STRICT_SUPPORT_MAX:
        status = "needs_review"   # 假从/从格临界，不直接改写喜忌
    else:
        status = "confirmed"

    return {
        "type": "从格", "sub_type": subtype, "status": status,
        "dominant_power": dominant_power,
        "is_fake": status == "needs_review",
        "supports": supports,
        "conflicts": conflicts,
        "evidence": base_evidence + [f"{k}={v}" for k, v in family_power.items()],
    }

14.2 专旺格判定

def check_zhuan_wang(chart: dict, strength: StrengthResult,
                     element_stats: dict, occurrences: list,
                     relations: list, kong_wang: dict) -> dict:
    """
    五类专旺格分别验证：曲直、炎上、稼穑、从革、润下。
    “日主极旺”只是入口条件，还须得月令或三会/三合之势、日主同五行成主势，
    并无有力官杀逆克。不得见身旺就直接返回专旺格。
    """
    day_stem = chart["day_stem"]
    day_element = STEM_ELEMENT[day_stem]
    month_branch = chart["month_branch"]
    config = {
        "木": {"name": "曲直格", "season": {"寅", "卯", "辰"}, "meeting": {"寅", "卯", "辰"}},
        "火": {"name": "炎上格", "season": {"巳", "午", "未"}, "meeting": {"巳", "午", "未"}},
        "土": {"name": "稼穑格", "season": {"辰", "戌", "丑", "未"}, "meeting": {"辰", "戌", "丑", "未"}},
        "金": {"name": "从革格", "season": {"申", "酉", "戌"}, "meeting": {"申", "酉", "戌"}},
        "水": {"name": "润下格", "season": {"亥", "子", "丑"}, "meeting": {"亥", "子", "丑"}},
    }
    rule = config[day_element]
    same_percent = element_stats[day_element]["percent"]
    resource_element = SHENG_ME_MAP[day_element]
    support_percent = same_percent + element_stats[resource_element]["percent"]
    officer_power = effective_power(occurrences, {"正官", "七杀"})
    branches = set(chart["branches"])
    seasonal_support = month_branch in rule["season"]
    meeting_support = rule["meeting"].issubset(branches) or has_effective_combination(
        relations, target_element=day_element, kinds={"三会", "三合"}
    )

    supports = []
    conflicts = []
    if strength.score >= ZHUAN_WANG_SCORE_MIN:
        supports.append(f"日主极旺，强弱分={strength.score}")
    else:
        conflicts.append("日主未达专旺阈值")
    if seasonal_support or meeting_support:
        supports.append("得月令或成三会/三合之势")
    else:
        conflicts.append("不得月令，亦无三会/三合成势")
    if support_percent >= ZHUAN_WANG_DOMINANCE_PERCENT:
        supports.append(f"同党与印星占比={support_percent}%")
    else:
        conflicts.append("日主同党与印星未成压倒性主势")
    if officer_power > ZHUAN_WANG_OFFICER_MAX:
        conflicts.append(f"官杀有力逆克，有效力量={officer_power}")

    if conflicts:
        status = "excluded"
    elif (strength.score < ZHUAN_WANG_STRICT_SCORE or
          support_percent < ZHUAN_WANG_STRICT_PERCENT):
        status = "needs_review"
    else:
        status = "confirmed"

    return {
        "type": "专旺格",
        "sub_type": rule["name"],
        "element": day_element,
        "status": status,
        "supports": supports,
        "conflicts": conflicts,
        "evidence": [
            f"月支={month_branch}", f"同印占比={support_percent}%",
            f"官杀有效力量={officer_power}",
        ],
    }

14.3 化气格判定

def check_hua_qi(chart: dict, relations: dict, kong_wang: dict) -> Optional[dict]:
    """
    五合化气判定：
    甲己→土, 乙庚→金, 丙辛→水, 丁壬→木, 戊癸→火
    """
    day_stem = chart["day_stem"]
    four_stems = chart["stems"]
    four_branches = chart["branches"]
    month_branch = chart["month_branch"]
    HUA_MAP = {
        ("甲","己"): ("土", "辰戌丑未"),
        ("乙","庚"): ("金", "申酉"),
        ("丙","辛"): ("水", "亥子"),
        ("丁","壬"): ("木", "寅卯"),
        ("戊","癸"): ("火", "巳午"),
    }
    
    # 日干只与紧贴的月干或时干先论化气，年干隔位不直接作为化气伴侣。
    for partner, (hua_element, hua_month) in HUA_MAP.items():
        if day_stem in partner:
            other = partner[0] if day_stem == partner[1] else partner[1]
            adjacent_positions = [i for i in (1, 3) if four_stems[i] == other]
            if adjacent_positions:
                # 条件1：月令是否为化神旺地
                month_ok = month_branch in hua_month or \
                           HIDDEN_STEMS[month_branch][0] in ELEMENT_STEMS[hua_element]
                
                # 条件2：地支是否有化神之根/局
                has_root = check_effective_element_root(
                    hua_element, four_branches, relations, kong_wang
                )
                
                # 条件3：无争合
                no_zheng_he = count_stem(four_stems, day_stem) == 1 and \
                              count_stem(four_stems, other) == 1
                
                # 条件4：无克破化神
                no_ke = not has_effective_counter_force(
                    hua_element, chart, relations, exclude_stems=partner
                )

                checks = {
                    "日干与化合干紧贴": True,
                    "月令引化": month_ok,
                    "化神有根或成局": has_root,
                    "无争合妒合": no_zheng_he,
                    "无有力克破化神": no_ke,
                }
                failed = [name for name, ok in checks.items() if not ok]
                return {
                    "type": "化气格",
                    "sub_type": f"{day_stem}{other}化{hua_element}格",
                    "hua_element": hua_element,
                    "pair": f"{day_stem}{other}合",
                    "status": "confirmed" if not failed else "excluded",
                    "is_true": not failed,
                    "supports": [name for name, ok in checks.items() if ok],
                    "conflicts": failed,
                    "note": "合而不化" if failed else "化气条件齐备",
                }
    
    return None

14.4 特殊格统一入口

```python
def evaluate_special_patterns(chart, strength, capacity, balance,
                              occurrences, relations, kong_wang):
    results = [
        check_cong_ge(chart, strength, occurrences, relations, kong_wang),
        check_zhuan_wang(chart, strength, balance, occurrences, relations, kong_wang),
    ]
    hua_qi = check_hua_qi(chart, relations, kong_wang)
    if hua_qi:
        results.append(hua_qi)

    # 同一命局若出现多个 confirmed，不得按数组顺序任取一个；
    # 应全部降为 needs_review，并输出格局冲突证据。
    confirmed = [r for r in results if r and r["status"] == "confirmed"]
    if len(confirmed) > 1:
        for r in confirmed:
            r["status"] = "needs_review"
            r.setdefault("conflicts", []).append("多特殊格同时通过，规则冲突待复核")
    return [r for r in results if r]
```

Step 15: 合冲刑害检测引擎

Step 16: 十神组合关系检测

十神组合不能用“只要在任意藏干中出现就成立”的布尔判断。每个十神出现项必须保留具体天干、透藏层级、柱位、根气、是否空亡、是否受冲合以及有效力量；日干自身是参照物，不得重复算成一个“比肩出现”。

```python
@dataclass
class TenGodOccurrence:
    stem: str
    ten_god: str
    position: str              # 年干/月干/日支本气等
    exposed: bool
    hidden_level: str | None   # 本气/中气/余气
    raw_power: float
    effective_power: float      # 已经计入月令、透藏、空亡、合冲损益
    kong_wang: bool
    damaged: bool

def can_form_combo(a, b):
    # 至少一方透干；双方均深藏且无引动时只记录潜在结构，不成立组合。
    return ((a.exposed or b.exposed)
            and a.effective_power >= COMBO_MIN_POWER
            and b.effective_power >= COMBO_MIN_POWER
            and not (a.damaged and b.damaged)
            and positions_can_interact(a.position, b.position))

def any_combo(group_a, group_b):
    return any(can_form_combo(a, b) for a in group_a for b in group_b)
```

17.1 主决策流程

def determine_yong_shen(
    chart: dict,
    strength: StrengthResult,
    climate: dict,
    pattern: dict,
    special_patterns: List[dict],
    combos: List[dict],       # 十神组合
    balance: dict,
    capacity: CarryingCapacity
) -> dict:
    """
    返回: {
        "yong_shen": str,      # 用神五行
        "xi_shen": str,        # 喜神五行
        "ji_shen": str,        # 忌神五行
        "chou_shen": str,      # 仇神五行
        "xian_shen": str,      # 闲神五行
        "method": str,         # 取用方法
        "reason": str,         # 理由
        "verification": dict   # 验证结果
    }
    """
    day_stem = chart["day_stem"]
    four_stems = chart["stems"]
    four_branches = chart["branches"]
    day_element = STEM_ELEMENT[day_stem]
    
    # ═══ 第一优先：仅已确认的特殊格局可锁定结论 ═══
    confirmed_special = select_confirmed_special_pattern(special_patterns)
    if confirmed_special:
        return _yong_shen_special(confirmed_special, day_element)
    
    # ═══ 第二优先：格局病药与相神 ═══
    disease = _find_disease(pattern, combos)
    candidates = collect_pattern_candidates(pattern, disease, chart)

    # ═══ 第三优先：调候约束（不是无条件覆盖格局） ═══
    if climate["needed"]:
        if climate["state"] == "missing":
            add_climate_candidates(candidates, climate, score=40)
        elif climate["state"] == "insufficient":
            add_climate_candidates(candidates, climate,
                                   score=interpolate_climate_score(climate, 5, 25))
        elif climate["state"] == "satisfied":
            add_evidence(candidates, "调候已备，不重复加分")
        elif climate["state"] == "excessive":
            penalize_same_element_candidates(candidates, climate["element"])
    
    # ═══ 第四优先：扶抑 ═══
    if strength.level in (Strength.EXTREME_STRONG, Strength.STRONG):
        yong = _yong_shen_strong(day_stem, strength, four_stems, four_branches)
    elif strength.level in (Strength.EXTREME_WEAK, Strength.WEAK):
        yong = _yong_shen_weak(day_stem, strength, four_stems, four_branches)
    else:
        # 中和 → 看格局与流通
        yong = _yong_shen_neutral(pattern, combos)
    
    # ═══ 第五优先：通关检查 ═══
    conflict = _check_two_element_conflict(four_stems, four_branches)
    if conflict:
        tong_guan = conflict["bridge_element"]
        # 通关优先于扶抑（若冲突严重）
        if conflict["severity"] == "high":
            yong = tong_guan
    
    candidates.extend(collect_fuyi_tongguan_candidates(
        yong, chart, strength, balance, capacity
    ))
    return _build_result_from_candidates(candidates, chart, pattern, climate,
                                         balance, capacity)

17.2 日主承载能力修正（候选天干级）

承载能力不是“再算一次身强身弱”，而是将日主强弱、根气稳定性、寒暖燥湿和流通情况合成一个仅供候选取用阶段使用的 0~100 指标。它的目的，是把“理论生克方向”变成“在此局里可否兑现”的有效分数。

```python
@dataclass
class CapacityConfig:
    clash_root_multiplier: float = 0.50
    low_capacity_threshold: float = 40.0
    review_score_gap: float = 5.0

def assess_carrying_capacity(chart, strength, climate, relations, balance, config):
    day = chart.day_gan
    day_element = get_wu_xing(day)

    # 根气按位置和有效性计分；被冲或被有效合化的根不按满分算。
    roots = score_effective_roots(
        day_element, chart.branches, relations,
        main_root=1.0, middle_root=0.6, residual_root=0.3,
        clash_multiplier=config.clash_root_multiplier  # 默认 0.5，不直接归零
    )  # 0..100
    climate_workability = score_day_master_workability(
        day_element, chart.month_branch, climate, balance
    )  # 0..100
    circulation = score_day_master_circulation(day_element, balance, relations)  # 0..100

    overall = clamp(
        0.40 * strength.score + 0.30 * roots.value
        + 0.20 * climate_workability.value + 0.10 * circulation.value,
        0, 100
    )
    return CarryingCapacity(
        overall=round(overall, 2), root_stability=roots.value,
        climate_workability=climate_workability.value,
        circulation=circulation.value,
        penalties=[*roots.penalties, *climate_workability.penalties,
                   *circulation.penalties],
        evidence=[*roots.evidence, *climate_workability.evidence,
                  *circulation.evidence]
    )
```

当 `capacity.overall < 40`，或首、次候选的 `effective_score` 差距小于 5 分时，输出 `status="needs_review"`，而不是输出“没有用神”。此时仍返回完整候选列表、折减证据和调候候选，供人工判断。

候选先根据“天干 + 作用类型”计算理论分，再决定是否需要承载修正；不要对所有候选统一乘系数：

```python
class CandidateEffect(Enum):
    CLIMATE_WARMTH = auto()       # 冬月丙火暖局：直接环境收益
    CLIMATE_MOISTURE = auto()     # 夏月壬水润局：直接环境收益
    RESOURCE_ACTIVATION = auto()  # 印星生身，需日主与环境能承接
    PEER_SUPPORT = auto()         # 比劫助身
    CONTROL_WEALTH = auto()       # 身强任财，需日主有驾驭能力
    OUTPUT_DRAIN = auto()         # 食伤泄秀，需日主有可泄之力

def apply_carrying_capacity(candidate, capacity):
    direct = sum(e.score for e in candidate.effects
                 if e.kind in (CLIMATE_WARMTH, CLIMATE_MOISTURE))
    active = sum(e.score for e in candidate.effects
                 if e.kind not in (CLIMATE_WARMTH, CLIMATE_MOISTURE))
    factor = clamp(capacity.overall / 100.0, 0.20, 1.00)
    candidate.raw_score = direct + active
    candidate.effective_score = direct + active * factor
    if active and factor < 1.0:
        candidate.conflicts.append(Evidence(
            rule="useful_god.carrying_capacity",
            subject=candidate.stem,
            relation="承载不足",
            points=candidate.effective_score - candidate.raw_score,
            reason=f"日主承载能力 {capacity.overall:.1f}/100，需日主参与的作用按 {factor:.2f} 折算"
        ))
    return candidate
```

17.4 喜忌推导（应用承载能力后排序）

新实现必须由 `score_stem_candidate(stem, ...)` 产生天干级候选；壬、癸虽然同属水，但调候次序、透藏根气和实际效力分开计算。最终排序使用 `effective_score`，并以 `raw_score`、规则优先级、证据数量作为同分决胜依据。

def _build_result_from_candidates(candidates: list, chart: dict, pattern: dict,
                                  climate: dict, balance: dict,
                                  capacity: CarryingCapacity) -> dict:
    """分角色重新评分，并强制五行角色互斥；喜神不等于“生用神者”。"""
    yong = choose_highest_candidate(candidates, key="effective_score")
    used_elements = {yong.element}

    xi_candidates = [score_candidate_for_role(c, "喜神", yong=yong,
                                               chart=chart, pattern=pattern,
                                               climate=climate, balance=balance)
                     for c in candidates if c.element not in used_elements]
    xi = choose_highest_candidate(xi_candidates, key="effective_score")
    used_elements.add(xi.element)

    ji_candidates = [score_candidate_for_role(c, "忌神", yong=yong,
                                               chart=chart, pattern=pattern,
                                               climate=climate, balance=balance)
                     for c in candidates if c.element not in used_elements]
    ji = choose_highest_candidate(ji_candidates, key="effective_score")
    used_elements.add(ji.element)

    chou_candidates = [score_candidate_for_role(c, "仇神", yong=yong, ji=ji,
                                                 chart=chart, pattern=pattern,
                                                 climate=climate, balance=balance)
                       for c in candidates if c.element not in used_elements]
    chou = choose_highest_candidate(chou_candidates, key="effective_score")
    used_elements.add(chou.element)
    xian_elements = sorted(set(FIVE_ELEMENTS) - used_elements)

    # 每个角色内的 percent = 该候选有效分 / 本角色所有正分候选总和。
    normalize_role_percent(candidates, role="用神")
    normalize_role_percent(xi_candidates, role="喜神")
    normalize_role_percent(ji_candidates, role="忌神")
    normalize_role_percent(chou_candidates, role="仇神")

    selected = {
        "yong_shen": yong,
        "xi_shen": xi,
        "ji_shen": ji,
        "chou_shen": chou,
        "xian_shen": xian_elements[0] if xian_elements else None,
    }
    assert len({yong.element, xi.element, ji.element, chou.element}) == 4
    return {
        "selected": selected,
        "candidates": candidates,
        "status": "needs_review" if (
            capacity.overall < LOW_CAPACITY_THRESHOLD or
            score_gap(candidates) < REVIEW_SCORE_GAP
        ) else "confirmed",
        "evidence": [*yong.reasons, *xi.reasons, *ji.reasons, *chou.reasons],
    }

17.5 用神验证（根气、关系、承载能力）

def verify_yong_shen(yong_stem: str, four_stems: List[str],
                     four_branches: List[str], clashes: List,
                     capacity: CarryingCapacity, kong_wang: dict) -> dict:
    """
    ① 是否有根  ② 是否被冲  ③ 是否被合化 ④ 是否有承载能力
    """
    # ① 有根检查
    yong_element = STEM_ELEMENT[yong_stem]
    exact_root_level = "无根"
    same_element_support = []
    for b in four_branches:
        for idx, h in enumerate(HIDDEN_STEMS[b]):
            if h == yong_stem:
                if idx == 0:
                    exact_root_level = "本气强根"
                elif idx == 1:
                    exact_root_level = "中气中根"
                else:
                    exact_root_level = "余气弱根"
            elif STEM_ELEMENT[h] == yong_element:
                same_element_support.append({"branch": b, "stem": h, "note": "同五行助力，不冒充本干根"})
    
    # 透干检查
    tou_gan = yong_stem in four_stems
    
    # ② 被冲检查
    is_chonged = False
    for b in four_branches:
        for h in HIDDEN_STEMS[b]:
            if h == yong_stem:
                if any(b in pair for pair in clashes):
                    is_chonged = True
    
    # ③ 被合检查
    is_he = False
    # 检查用神对应天干是否被五合
    for ys in [s for s in four_stems if s == yong_stem]:
        for pair in TIAN_GAN_HE:
            if ys in pair:
                other = pair[0] if ys == pair[1] else pair[1]
                if other in four_stems:
                    is_he = True
    
    return {
        "stem": yong_stem,
        "element": yong_element,
        "exact_root": exact_root_level,
        "same_element_support": same_element_support,
        "tou_gan": tou_gan,
        "is_chonged": is_chonged,
        "is_he": is_he,
        "kong_wang_affected": exact_root_is_kong_wang(yong_stem, four_branches, kong_wang),
        "effective": exact_root_level != "无根" and not is_chonged,
        "power": _calc_power(exact_root_level, tou_gan, is_chonged, is_he, kong_wang),
        "carrying_capacity": capacity.overall,
        "capacity_evidence": capacity.evidence
    }

Step 18: 生克链连贯性分析

def analyze_sheng_ke_chain(four_stems, four_branches, day_stem) -> dict:
    """
    检查五行流通路径是否连贯
    印→身→食伤→财→官→印
    """
    day_element = STEM_ELEMENT[day_stem]
    
    # 构建五行力量表
    power = calc_five_element_power(four_stems, four_branches)
    
    # 定义链条：印→身→食伤→财→官→(印)
    chain = [
        SHENG_ME_MAP[day_element],  # 印
        day_element,                 # 身
        SHENG_MAP[day_element],      # 食伤
        KE_MAP[day_element],         # 财
        KE_ME_MAP[day_element],      # 官
    ]
    
    breaks = []
    for i in range(len(chain) - 1):
        current = chain[i]
        next_el = chain[i + 1]
        
        # 检查current是否有力
        if power.get(current, 0) < 5:  # 阈值可调
            breaks.append({
                "position": f"{current}→{next_el}",
                "reason": f"{current}太弱(力量{power.get(current,0)})，链断",
                "severity": "high"
            })
    
    return {
        "chain": chain,
        "power": power,
        "breaks": breaks,
        "is_smooth": len(breaks) == 0,
        "disease_point": breaks[0]["position"] if breaks else None
    }

Step 19: 大运、流年、流月联合预判

【算法口径：先逐层评价具体干支，再评价岁运层之间的组合；十神、承载、刑冲合害、空亡和制化均保留证据，不以单一总分代替命理判断】

class TransitContext:
    dayun: Optional[Pillar]
    liunian: Optional[Pillar]
    liuyue: Optional[Pillar]

class TransitAnalysis:
    context: TransitContext
    layer_impacts: list[FortuneImpact]       # 每一层自身对原局的作用
    interactions: list[TransitInteraction]   # 层间及岁运与原局的组合
    projected_strength: float                # 岁运加入后的强弱估计，0～100
    requires_reselection: bool               # 是否跨强弱阈值而须重新取用
    total_score: float                       # 仅作结构排序，不是吉凶概率
    risk: str                                # 平稳、留意、预警、重点预警
    review_notes: list[str]

def analyze_transit(context: TransitContext, original: AnalysisResult) -> TransitAnalysis:
    layers = compact([
        ("大运", context.dayun, 1.00),
        ("流年", context.liunian, 0.65),
        ("流月", context.liuyue, 0.35),
    ])

    # 一、逐层评价：必须按具体天干和运支本气的具体十神分论。
    # 庚正官与辛七杀、癸偏印与壬正印、丁食神与丙伤官不得因同五行而共用一分。
    for label, pillar, time_weight in layers:
        main_stem = hidden_stems(pillar.branch)[0]
        strength = stem_branch_capacity(pillar.stem, pillar.branch)
        # 承载修正同时看通根、十二长生和盖头截脚：
        # 临官/帝旺1.25，长生/冠带1.15，墓库1.05，衰/病0.80，死/绝0.60；
        # 其余再按坐支本气对运干的生克泄耗修正。
        # 因此甲申处绝地须折减，丙戌处墓库为有根而闭藏，不能都按普通火土处理。
        impact = score_exact_stem_and_ten_god(
            arriving_stem=pillar.stem,
            branch_main_stem=main_stem,
            selected_roles=original.useful_gods,
            route=original.useful_gods.route,
            stem_strength=strength,
        )

        # 七杀先论攻身，再查食神制杀、印化杀；“有救”不得抹去“先伤”。
        if impact.has("七杀"):
            impact += score_killer_rescue(
                food_power=effective_power("食神", original, pillar),
                seal_power=effective_power({"正印", "偏印"}, original, pillar),
            )

        # 空亡必须以原局旬空为参照。岁运值原局空支称“填实”，只标应期，
        # 吉凶沿用该支本气十神与制化结果，不重复计分。
        if pillar.branch in original.kong_wang.branches:
            impact.add_marker(classify_filled_kong_wang(main_stem))
        # “某干支相对自己的旬为空”在定义上不成立，禁止据此机械减半。

        # 冲按宫位分层，月令提纲与日支不能降格为普通根气。
        for position, natal_pillar in enumerate(original.input_pillars):
            if is_clash(pillar.branch, natal_pillar.branch):
                if position == MONTH:
                    impact.add("冲提纲", -12, no_duplicate_root_score=True)
                elif position == DAY:
                    impact.add("冲日支", -6, no_duplicate_ji_root_bonus=True)
                elif contains_root(natal_pillar.branch, original.useful_gods.yong_shen):
                    impact.add("冲用神根", -12)
                    if original.kong_wang.affects(position):
                        impact.review("冲空有冲起、冲实等异说；当前从严折减")
                elif contains_root(natal_pillar.branch, original.useful_gods.ji_shen):
                    impact.add("冲忌神根", +8)

        # 天干五合按“合住谁”判作用，不能只有“合用神”一条。
        for natal_pillar in original.input_pillars:
            if is_stem_combine(pillar.stem, natal_pillar.stem):
                impact += score_stem_combine_target(
                    target=natal_pillar.stem,
                    yong=-6, xi=-5, ji=+6, chou=+4, climate=-8,
                )
                if other_stem_also_combines(natal_pillar.stem):
                    impact.add("争合", -6)
                # 是否化气必须另验月令、透根和受冲，不因相合直接认化。

        # 运支补齐三合、三会时按成局后五行评价，并复核成化条件。
        impact += score_completed_branch_groups(original.branches, pillar.branch)

        # 地支六合由基础层统一返回合化方向；“相合”不等于“已经合化”。
        for position, natal_branch in enumerate(original.branches):
            if combine_element := branch_combine_element(pillar.branch, natal_branch):
                interactions.add(
                    "地支六合",
                    score_combine_direction(combine_element,
                                            yong=+4, xi=+2.5, ji=-4, chou=-3),
                    note="只按合势归向评价，仍须验证月令、透干及冲刑破害",
                )
                if position == MONTH:
                    interactions.add("六合提纲", -4,
                                     note="提纲受牵绊、事情胶着，不被归向吉分完全抵销")
                if natal_branch.contains_exact_root(original.yong_stem):
                    interactions.add("合绊用神根", -6,
                                     note="候选本干根气受牵绊，合化是否成立另验")
                if pillar.branch in original.kong_wang.branches:
                    interactions.note("参与六合的岁运支同时填实原局旬空，不重复计分")

        # 伤官直犯正官用神为组合级风险，地支喜神不得完全抵销。
        if ten_god(pillar.stem) == "伤官" and ten_god(original.yong_stem) == "正官":
            impact.add("伤官见官", -10)

        layer_impacts.append(impact)

    # 二、岁运与原局、岁运层之间的柱级组合。
    for layer in layers:
        for natal_pillar in original.input_pillars:
            if layer.pillar == natal_pillar:
                interactions.add("伏吟", -8)
            if stems_mutually_control(layer.pillar, natal_pillar) and \
               branches_clash(layer.pillar, natal_pillar):
                interactions.add("天克地冲（反吟级结构）", -14)

    for left, right in combinations(layers, 2):
        if left.pillar == right.pillar:
            interactions.add("岁运并临", -10)
        if stems_mutually_control(left.pillar, right.pillar) and \
           branches_clash(left.pillar, right.pillar):
            interactions.add("岁运相战", -12)

    # 三、刑必须在“原局全部地支 + 所有岁运地支”的联合集合中判断。
    all_branches = original.branches + [layer.pillar.branch for layer in layers]
    for branch in {"辰", "午", "酉", "亥"}:
        if count(all_branches, branch) >= 2:
            interactions.add(f"{branch}{branch}自刑", -6)
    if contains_all(all_branches, {"寅", "巳", "申"}):
        interactions.add("寅巳申无恩之刑", -12)
    if contains_all(all_branches, {"丑", "戌", "未"}):
        interactions.add("丑戌未恃势之刑", -12, review_month_command=True)

    # 四、动态制化须检查原局与各岁运层共同到位的十神。
    active_gods = exact_ten_gods_of_stems_and_branch_main_qi(original, layers)
    if active_gods.has("七杀", "食神"):
        interactions.add("食神制杀", +8, note="有救而先伤")
    elif active_gods.has("七杀", {"正印", "偏印"}):
        interactions.add("印化杀", +6, note="有救而先伤")
    if active_gods.has("伤官", {"正印", "偏印"}):
        interactions.add("伤官配印", +6)
    if active_gods.has({"正财", "偏财"}) and original.climate.uses_water:
        interactions.add("财坏印并伤调候", -6)

    # 五、动态取用：大运定基调，流年定应期，流月只作短期触发。
    projected_strength = clamp(
        original.strength.final_score +
        sum(layer_strength_delta(layer) * layer.time_weight for layer in layers),
        0, 100,
    )
    requires_reselection = strength_band(projected_strength) != \
                           strength_band(original.strength.final_score)
    if requires_reselection:
        interactions.add("重新取用复核", 0,
                         note="强弱跨阈值，原局静态喜忌不得机械沿用")

    total_score = sum(x.score for x in layer_impacts) + \
                  sum(x.score for x in interactions)
    risk = classify_transit_risk(total_score, interactions)
    return TransitAnalysis(context, layer_impacts, interactions,
                           projected_strength, requires_reselection,
                           total_score, risk, collect_review_notes())

def full_analysis(birth_data: dict) -> dict:
    """

    relations = detect_all_relations(chart)

    balance = calc_element_balance(chart, all_relations, DEFAULT_CONFIG.elements)

    # 12. 调候

    chain = analyze_sheng_ke_chain(chart["stems"], chart["branches"], chart["day_stem"])
